# 第 08 章：命令历史（↑↓ 翻，steps/step08_history.py）

## 本步目标

在上一章美化好的界面上，输入框现在有了记忆：发出去的话会被记下来，按 `↑` 逐条翻出历史消息，按 `↓` 往回翻，翻到底还能恢复你刚才写到一半的内容——和 bash、Python REPL 里的体验一样。和上一章的区别只有输入框：`Input` 换成了自己写的 `HistoryInput`，其余代码原封不动。

## 新概念

- **`on_key` 事件**：Widget 上的按键回调。每次有按键落到这个组件上（焦点在它身上时），Textual 都会调用 `on_key(event)`，`event.key` 是按键名（`"up"`、`"down"`、`"enter"`、单个字符等）。这是做「自定义按键行为」的入口。
- **`event.stop()`**：阻止事件继续传播。Textual 里按键事件会从焦点组件一路往上传给父组件，直到有人处理。如果不在 `HistoryInput` 里拦住 `↑`，它还会传到外层的滚动区，被当成「向上滚屏」——按一下就既翻了历史又滚了屏。`stop()` 就是「这个按键我处理了，到此为止」。
- **历史 + 游标 + 草稿三件套**：实现命令历史的经典数据结构。`_history` 列表存所有发过的内容；`_history_pos` 是游标，记录当前翻到了第几条（`None` 表示「没在翻历史，正在编辑新内容」）；`_draft` 是开始翻历史之前输入框里的内容，翻到底时还原，保证打到一半的字不丢。

构造函数：在父类基础上加三个状态。

```python
class HistoryInput(Input):
    """带上下历史记忆的输入框。"""

    def __init__(self, *args, **kwargs) -> None:
        super().__init__(*args, **kwargs)
        self._history: list[str] = []
        # 当前翻到历史的第几条；None 表示没在翻历史（正在编辑新内容）
        self._history_pos: int | None = None
        # 开始翻历史之前，输入框里的内容（草稿）
        self._draft: str = ""
```

`on_key`：只接管 `↑` 和 `↓`，其余按键放行。

```python
    async def on_key(self, event: events.Key) -> None:
        if event.key == "up":
            event.stop()  # 阻止按键继续传播（比如被滚动区当成滚屏）
            self._history_previous()
        elif event.key == "down":
            event.stop()
            self._history_next()
        else:
            # 用户开始打字了，就退出「翻历史」状态
            self._history_pos = None
```

注意 `else` 分支：翻到某条历史后如果用户直接开始打字，游标要重置回 `None`。不然下次按 `↓` 会从旧游标接着翻，行为会很怪。

`↑` 的处理：第一次按先存草稿。

```python
    def _history_previous(self) -> None:
        """↑ ：往更早的历史翻。"""
        if not self._history:
            return
        if self._history_pos is None:
            # 第一次按 ↑：先保存草稿，再跳到最新一条历史
            self._draft = self.value
            self._history_pos = len(self._history) - 1
        elif self._history_pos > 0:
            self._history_pos -= 1
        self.value = self._history[self._history_pos]
```

三个分支：没有历史就什么都不做；第一次在翻（`_history_pos is None`）就把当前输入存进 `_draft`，游标指向最新一条；否则游标往前挪一格。挪到 `0` 就不再往前——一直按 `↑` 会停在最早一条上。

`↓` 的处理和发送后记录：

```python
    def _history_next(self) -> None:
        """↓ ：往更近的历史翻，翻过头就恢复草稿。"""
        if self._history_pos is None:
            return
        self._history_pos += 1
        if self._history_pos >= len(self._history):
            self._history_pos = None
            self.value = self._draft     # 翻过头：还原草稿，退出翻历史状态
        else:
            self.value = self._history[self._history_pos]

    def push_history(self, text: str) -> None:
        """发送成功后，把这条内容记入历史。"""
        if text:
            self._history.append(text)
        self._history_pos = None
        self._draft = ""
```

`↓` 翻过头（游标越过最后一条）时恢复草稿并把游标归 `None`。`push_history` 在每次发送成功后由 App 调用：记下这条内容，同时清空游标和草稿，为下一轮做准备。

```python
THINKING = "Simple greeting, respond in Chinese."

class HistoryInput(Input):
    """带上下历史记忆的输入框。"""

    def __init__(self, *args, **kwargs) -> None:
        super().__init__(*args, **kwargs)
        self._history: list[str] = []
        # 当前翻到历史的第几条；None 表示没在翻历史（正在编辑新内容）
        self._history_pos: int | None = None
        # 开始翻历史之前，输入框里的内容（草稿）
        self._draft: str = ""

    async def on_key(self, event: events.Key) -> None:
        if event.key == "up":
            event.stop()  # 阻止按键继续传播（比如被滚动区当成滚屏）
            self._history_previous()
        elif event.key == "down":
            event.stop()
            self._history_next()
        else:
            # 用户开始打字了，就退出「翻历史」状态
            self._history_pos = None

    def _history_previous(self) -> None:
        """↑ ：往更早的历史翻。"""
        if not self._history:
            return
        if self._history_pos is None:
            # 第一次按 ↑：先保存草稿，再跳到最新一条历史
            self._draft = self.value
            self._history_pos = len(self._history) - 1
        elif self._history_pos > 0:
            self._history_pos -= 1
        self.value = self._history[self._history_pos]

    def _history_next(self) -> None:
        """↓ ：往更近的历史翻，翻过头就恢复草稿。"""
        if self._history_pos is None:
            return
        self._history_pos += 1
        if self._history_pos >= len(self._history):
            self._history_pos = None
            self.value = self._draft
        else:
            self.value = self._history[self._history_pos]

    def push_history(self, text: str) -> None:
        """发送成功后，把这条内容记入历史。"""
        if text:
            self._history.append(text)
        self._history_pos = None
        self._draft = ""



class ChatScroll(VerticalScroll):
    """聊天滚动区：修复 anchor() 的一个小 bug。

    anchor() 让滚动区一直吸底；但 Textual 的合成器在「内容不足一屏」时
    会把 scroll_y 直接设成负数（set_reactive 绕过了 0 下限的校验），表现为：
    第一次发消息后，上面的内容整体往下挪、顶部空出一片。
    这里把负的滚动值挡回去即可。
    """

    def set_reactive(self, reactive, value) -> None:
        if (
            isinstance(value, (int, float))
            and value < 0
            and (reactive is Widget.scroll_y or reactive is Widget.scroll_target_y)
        ):
            value = 0
        super().set_reactive(reactive, value)


class ChatApp(App):
    CSS = """
    Screen {
        layout: vertical;
    }
    #chat-scroll {
        width: 1fr;
        height: 1fr;
        padding: 0;
    }
    #input-area {
        dock: bottom;
        width: 1fr;
        height: auto;
    }
    #input-row {
        width: 1fr;
        height: 3;
        border: round #3a3f4a;
        padding: 0 1;
        margin: 1 1 0 1;
    }
    #prompt {
        width: auto;
        height: auto;
        color: #c8cdd5;
        padding-right: 1;
    }
    #user-input {
        width: 1fr;
        height: auto;
        border: none;
        padding: 0;
    }
    """

    def compose(self) -> ComposeResult:
        with ChatScroll(id="chat-scroll"):
            yield WelcomeBox()
        with Vertical(id="input-area"):
            with Horizontal(id="input-row"):
                yield Static(">", id="prompt")
                # 唯一的区别：这里从 Input 换成了 HistoryInput
                yield HistoryInput(placeholder="", id="user-input")

    def on_mount(self) -> None:
        """启动后自动聚焦输入框。"""
        self.query_one("#user-input", Input).focus()

    async def on_input_submitted(self, event: Input.Submitted) -> None:
        user_text = event.value.strip()
        if not user_text:
            return

        input_widget = self.query_one("#user-input", HistoryInput)
        input_widget.value = ""
        input_widget.push_history(user_text)  # 记入历史

        scroll = self.query_one("#chat-scroll", VerticalScroll)
        await scroll.mount(UserMessage(user_text))
        message = AssistantMessage()
        await scroll.mount(message)

        scroll.anchor()

        self.run_worker(self.stream_response(message))

    async def stream_response(self, message: AssistantMessage) -> None:
        """先逐字写出思考过程，再流式输出回答。"""
        thinking_content = message.query_one("#thinking-content", Static)
        thinking_row = message.query_one(".thinking-row")
        buffer = ""
        for char in THINKING:
            if not buffer:
                thinking_row.remove_class("stream-pending")
            buffer += char
            thinking_content.update(buffer)
            await asyncio.sleep(0.02)

        md = message.query_one("#answer-md", Markdown)
        answer_row = message.query_one(".answer-row")
        stream = Markdown.get_stream(md)
        try:
            for i in range(0, len(ANSWER), 8):
                if i == 0:
                    answer_row.remove_class("stream-pending")
                await stream.write(ANSWER[i : i + 8])
                await asyncio.sleep(0.05)
        finally:
            await stream.stop()


if __name__ == "__main__":
    app = ChatApp()
    app.run()
```

界面和第 07 章一模一样，差别全在输入框行为上。按下面的顺序戳一戳：

- 接着按 `↓` 回到「第二条」，再按 `↓` 输入框变空——这就是翻过头恢复草稿（草稿本来就是空的）。
- 验证草稿：先发一条「旧消息」，然后输入「写到一半」（别回车），按 `↑` 输入框变成「旧消息」，再按 `↓` 应该恢复成「写到一半」。
- 注意按 `↑` 时聊天区不应该跟着滚动——这就是 `event.stop()` 的作用。

上面四条手动验证其实已经覆盖了两个核心场景：

- **历史导航**：发两条消息后，`↑` 两次依次翻出「第二条」「第一条」，`↓` 两次回到「第二条」再翻到空草稿——覆盖了游标的全部状态转移。
- **草稿保留**：先存一条历史，再打到一半按 `↑` 翻走、按 `↓` 翻回，草稿「写到一半」完整恢复——这是草稿机制的生命线。

## 常见问题

- **按 `↑` 时聊天记录区跟着滚了**：`on_key` 里漏了 `event.stop()`。按键事件会沿组件树向上传播，滚动区也监听方向键当滚屏用。拦截到的按键必须 `stop()`。
- **翻历史后打到一半的字丢了**：第一次按 `↑` 之前没有把 `self.value` 存进 `_draft`，或者 `↓` 翻过头时忘了恢复。草稿是「翻历史」这个功能的体验底线，bash 也是这么做的。
- **翻到某条历史后直接打字，再按 `↓` 行为错乱**：漏了 `on_key` 的 `else` 分支。用户一旦按了 `↑`/`↓` 以外的键，就说明他在编辑了，要把 `_history_pos` 重置为 `None`，否则游标还停在旧位置，`↓` 会从那里继续翻。
